cmake_minimum_required(VERSION 3.21)
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_C_COMPILER armclang.exe)
set(CMAKE_CXX_COMPILER armclang.exe)
set(CMAKE_ASM_COMPILER armclang.exe)
set(CMAKE_C_COMPILER_WORKS TRUE)
set(CMAKE_CXX_COMPILER_WORKS TRUE)
set(CMAKE_ASM_COMPILER_WORKS TRUE)
project(FinsIntegrated C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 99)
set(LEGACY "${CMAKE_CURRENT_LIST_DIR}/../inertial-odometry")
set(BOARD "${LEGACY}/BSP/Robomaster_A")
set(PROTOCOL "${CMAKE_CURRENT_LIST_DIR}/../../protocol/generated/c")
set(FINS_DEPENDENCY_ROOT "${LEGACY}/ThirdParty" CACHE PATH "Pinned CMSIS-DSP and ETL checkout root")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_C_FLAGS "--target=arm-arm-none-eabi -mcpu=cortex-m4 -mfloat-abi=hard -mfpu=fpv4-sp-d16 -Os -funsigned-char -fshort-enums -fshort-wchar -ffunction-sections -gdwarf-4 -Wno-license-management")
set(CMAKE_CXX_FLAGS "${CMAKE_C_FLAGS} -fno-rtti")
set(CMAKE_ASM_FLAGS "--target=arm-arm-none-eabi -mcpu=cortex-m4 -masm=auto -c -gdwarf-4")
set(CMAKE_C_LINK_EXECUTABLE "armlink.exe <OBJECTS> <LINK_FLAGS> --output=<TARGET>")
set(CMAKE_CXX_LINK_EXECUTABLE "armlink.exe <OBJECTS> <LINK_FLAGS> --output=<TARGET>")
set(CMAKE_EXE_LINKER_FLAGS "--cpu=cortex-m4 --strict --scatter=${BOARD}/MDK-ARM/Robomaster_A/Robomaster_A.sct --info=sizes --map --list=${CMAKE_BINARY_DIR}/Robomaster_A.map")
add_compile_definitions(USE_HAL_DRIVER STM32F427xx __ROBOMASTER_A __CMAKE_APP ARM_MATH_CM4 ARM_MATH_MATRIX_CHECK ARM_MATH_ROUNDING)
option(FINS_DEPTH_CALIBRATED "Enable depth hold ONLY after verifying shared NX/MCU datum and sensor calibration" OFF)
set(FINS_DEPTH_DATUM_PA "101325.0f" CACHE STRING "Four-pressure mean zero datum, Pa; must match odometry")
add_compile_definitions(FINS_DEPTH_CALIBRATED=$<BOOL:${FINS_DEPTH_CALIBRATED}> FINS_DEPTH_DATUM_PA=${FINS_DEPTH_DATUM_PA})
include_directories(Services ${PROTOCOL})
foreach(part Interface Devices Algorithms Services Components)
  file(GLOB_RECURSE headers "${LEGACY}/${part}/*.h" "${LEGACY}/${part}/*.hpp")
  include_directories("${LEGACY}/${part}")
  foreach(header IN LISTS headers)
    get_filename_component(dir "${header}" DIRECTORY)
    include_directories("${dir}")
  endforeach()
  file(GLOB_RECURSE sources "${LEGACY}/${part}/*.c" "${LEGACY}/${part}/*.cpp")
  list(APPEND sources_all ${sources})
endforeach()
list(FILTER sources_all EXCLUDE REGEX "/Interface/Examples/TaskSUB.cpp$")
include_directories("${BOARD}" "${BOARD}/FineMote_BSP" "${BOARD}/Core/Inc"
  "${BOARD}/Drivers/CMSIS/Include" "${BOARD}/Drivers/CMSIS/Device/ST/STM32F4xx/Include"
  "${BOARD}/Drivers/STM32F4xx_HAL_Driver/Inc" "${BOARD}/Drivers/STM32F4xx_HAL_Driver/Inc/Legacy"
  "${FINS_DEPENDENCY_ROOT}/etl/include" "${FINS_DEPENDENCY_ROOT}/CMSIS-DSP/Include"
  "${FINS_DEPENDENCY_ROOT}/CMSIS-DSP/Include/dsp" "${FINS_DEPENDENCY_ROOT}/CMSIS-DSP/PrivateInclude")
if(NOT EXISTS "${FINS_DEPENDENCY_ROOT}/etl/include/etl/queue.h")
  message(FATAL_ERROR "Run git submodule update --init --recursive before building")
endif()
foreach(group BasicMath CommonTables ComplexMath Controller FastMath Matrix QuaternionMath Statistics Support)
  file(GLOB dsp "${FINS_DEPENDENCY_ROOT}/CMSIS-DSP/Source/${group}Functions/${group}Functions.c")
  list(APPEND sources_all ${dsp})
endforeach()
list(APPEND sources_all "${FINS_DEPENDENCY_ROOT}/CMSIS-DSP/Source/CommonTables/CommonTables.c")
file(GLOB_RECURSE board_sources "${BOARD}/Core/Src/*.c" "${BOARD}/Drivers/*.c")
file(GLOB board_cpp "${BOARD}/*.cpp")
file(GLOB startup "${BOARD}/MDK-ARM/startup_*.s")
list(FILTER board_sources EXCLUDE REGEX "/(syscalls|sysmem).c$")
add_executable(Robomaster_A ${sources_all} ${board_sources} ${board_cpp} ${startup} Interface/TaskIntegrated.cpp)
set_target_properties(Robomaster_A PROPERTIES SUFFIX ".elf")
add_custom_command(TARGET Robomaster_A POST_BUILD COMMAND fromelf.exe --i32combined --output=${CMAKE_BINARY_DIR}/bin/Robomaster_A.hex $<TARGET_FILE:Robomaster_A>)
